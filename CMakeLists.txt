cmake_minimum_required(VERSION 3.16)
project(FruitNinjaNativeReconstruction VERSION 0.1.0 LANGUAGES C CXX)
# The Windows wrapper explicitly configures before every build. Suppress
# MSBuild's per-project CMake checks to avoid concurrent regeneration when
# extracted sources carry timestamps newer than the local machine's clock.
if(WIN32)
 option(CMAKE_SUPPRESS_REGENERATION "Configure explicitly before building" ON)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(FRUIT_BUILD_GAME "Build native SDL application" ON)
option(FRUIT_BUILD_TESTS "Build deterministic gameplay checks" ON)
option(FRUIT_USE_SYSTEM_SDL "Use an installed SDL2 when available" ON)
option(FRUIT_STRICT_WARNINGS "Treat reconstruction code warnings as errors" OFF)
set(FRUIT_BUILD_INSPECTOR ON CACHE BOOL "Asset inspection CLI")
if(NINTENDO_3DS)
 set(N3DS TRUE)
endif()
if(EMSCRIPTEN OR VITA OR N3DS)
 set(FRUIT_BUILD_TESTS OFF)
 set(FRUIT_BUILD_INSPECTOR OFF CACHE BOOL "Asset inspection CLI" FORCE)
 set(FRUIT_USE_SYSTEM_SDL OFF)
endif()
if(EMSCRIPTEN)
 add_compile_options(-fexceptions)
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS" OR ANDROID)
 set(FRUIT_BUILD_TESTS OFF)
 set(FRUIT_BUILD_INSPECTOR OFF CACHE BOOL "Asset inspection CLI" FORCE)
 set(FRUIT_USE_SYSTEM_SDL OFF)
endif()
if(ANDROID)
 set(CMAKE_POSITION_INDEPENDENT_CODE ON)
endif()
add_subdirectory(portable)
add_library(fruit_xml STATIC vendor/tinyxml2/tinyxml2.cpp)
target_include_directories(fruit_xml PUBLIC vendor/tinyxml2)
add_library(fruit_core STATIC game/src/config.cpp game/src/game.cpp game/src/save.cpp game/src/progress.cpp game/src/combos.cpp)
target_include_directories(fruit_core PUBLIC game/include)
target_link_libraries(fruit_core PUBLIC fruit_xml)
function(fruit_warnings target)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4)
  if(FRUIT_STRICT_WARNINGS)
   target_compile_options(${target} PRIVATE /WX)
  endif()
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  if(FRUIT_STRICT_WARNINGS)
   target_compile_options(${target} PRIVATE -Werror)
  endif()
 endif()
endfunction()
fruit_warnings(fruit_core)
if(FRUIT_BUILD_GAME)
 if(EMSCRIPTEN)
  add_library(fruit_web_sdl INTERFACE)
  target_compile_options(fruit_web_sdl INTERFACE "-sUSE_SDL=2")
  target_link_options(fruit_web_sdl INTERFACE "-sUSE_SDL=2")
  add_library(SDL2::SDL2 ALIAS fruit_web_sdl)
 endif()
 if(FRUIT_USE_SYSTEM_SDL)
  find_package(SDL2 2.0.18 QUIET CONFIG)
  if(NOT TARGET SDL2::SDL2 AND UNIX AND NOT APPLE)
   find_file(FRUIT_SDL_RUNTIME NAMES libSDL2-2.0.so.0 PATHS /usr/lib/x86_64-linux-gnu /usr/lib/aarch64-linux-gnu /usr/lib /usr/local/lib)
   if(FRUIT_SDL_RUNTIME)
    add_library(SDL2::SDL2 SHARED IMPORTED)
    set_target_properties(SDL2::SDL2 PROPERTIES IMPORTED_LOCATION "${FRUIT_SDL_RUNTIME}" INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/vendor/SDL2/include")
   endif()
  endif()
 endif()
 if(NOT TARGET SDL2::SDL2)
  if(ANDROID)
   set(SDL_SHARED ON CACHE BOOL "" FORCE)
   set(SDL_STATIC OFF CACHE BOOL "" FORCE)
  else()
   set(SDL_SHARED OFF CACHE BOOL "" FORCE)
   set(SDL_STATIC ON CACHE BOOL "" FORCE)
  endif()
  set(SDL_TEST OFF CACHE BOOL "" FORCE)
  set(SDL_TESTS OFF CACHE BOOL "" FORCE)
  set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
  add_subdirectory(vendor/SDL2 EXCLUDE_FROM_ALL)
  if(TARGET SDL2-static AND NOT TARGET SDL2::SDL2)
   add_library(SDL2::SDL2 ALIAS SDL2-static)
  endif()
 endif()
 add_library(fruit_vorbis STATIC vendor/stb_vorbis.c)
 if(UNIX)
  target_link_libraries(fruit_vorbis PUBLIC m)
 endif()
 add_library(fruit_platform STATIC game/src/renderer.cpp game/src/audio.cpp game/src/music_stream.cpp)
 target_include_directories(fruit_platform PRIVATE vendor)
 target_link_libraries(fruit_platform PUBLIC fruit_core fruit_assets SDL2::SDL2 fruit_vorbis)
 fruit_warnings(fruit_platform)
 if(ANDROID)
  add_library(fruit_ninja SHARED game/src/main.cpp)
  set_target_properties(fruit_ninja PROPERTIES OUTPUT_NAME main)
 elseif(APPLE)
  add_executable(fruit_ninja MACOSX_BUNDLE game/src/main.cpp)
 else()
  add_executable(fruit_ninja game/src/main.cpp)
 endif()
 if(NOT ANDROID AND TARGET SDL2::SDL2main)
  target_link_libraries(fruit_ninja PRIVATE SDL2::SDL2main)
 elseif(NOT ANDROID AND TARGET SDL2main)
  target_link_libraries(fruit_ninja PRIVATE SDL2main)
 endif()
 target_link_libraries(fruit_ninja PRIVATE fruit_platform)
 fruit_warnings(fruit_ninja)
 if(APPLE)
  set_target_properties(fruit_ninja PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "org.fruitnative.reconstruction" MACOSX_BUNDLE_BUNDLE_NAME "Fruit Ninja Native" MACOSX_BUNDLE_BUNDLE_VERSION "1" MACOSX_BUNDLE_SHORT_VERSION_STRING "0.1.0")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
   set_target_properties(fruit_ninja PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/Info.plist" XCODE_ATTRIBUTE_SKIP_INSTALL "NO" XCODE_ATTRIBUTE_TARGETED_DEVICE_FAMILY "1,2" XCODE_ATTRIBUTE_IPHONEOS_DEPLOYMENT_TARGET "13.0")
  endif()
  set(FRUIT_RUNTIME_ASSETS "$<TARGET_BUNDLE_DIR:fruit_ninja>/Contents/Resources/assets")
  if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
   set(FRUIT_RUNTIME_ASSETS "$<TARGET_BUNDLE_DIR:fruit_ninja>/assets")
  endif()
 else()
  set(FRUIT_RUNTIME_ASSETS "$<TARGET_FILE_DIR:fruit_ninja>/assets")
 endif()
 if(IOS)
  # Xcode bundle resources survive archives and incremental builds. POST_BUILD
  # alone did not put game resources in the archived iOS application.
  file(GLOB_RECURSE FRUIT_IOS_RESOURCES CONFIGURE_DEPENDS
       "${CMAKE_CURRENT_SOURCE_DIR}/assets/original/*"
       "${CMAKE_CURRENT_SOURCE_DIR}/assets/config/*"
       "${CMAKE_CURRENT_SOURCE_DIR}/platform/ios/icons/*.png")
  foreach(resource IN LISTS FRUIT_IOS_RESOURCES)
   file(RELATIVE_PATH relative "${CMAKE_CURRENT_SOURCE_DIR}" "${resource}")
   if(relative MATCHES "^assets/")
    get_filename_component(destination "${relative}" DIRECTORY)
   else()
    set(destination ".")
   endif()
   set_source_files_properties("${resource}" PROPERTIES MACOSX_PACKAGE_LOCATION "${destination}")
  endforeach()
  target_sources(fruit_ninja PRIVATE ${FRUIT_IOS_RESOURCES})
 endif()
 if(NOT IOS AND NOT ANDROID AND NOT EMSCRIPTEN AND NOT VITA AND NOT N3DS)
 add_custom_command(TARGET fruit_ninja POST_BUILD COMMAND ${CMAKE_COMMAND} -E make_directory "${FRUIT_RUNTIME_ASSETS}" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/original" "${FRUIT_RUNTIME_ASSETS}/original" COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/assets/config" "${FRUIT_RUNTIME_ASSETS}/config" VERBATIM)
 endif()
 if(EMSCRIPTEN)
  include(platform/wasm/platform.cmake)
 elseif(VITA)
  include(platform/vita/platform.cmake)
 elseif(N3DS)
  include(platform/3ds/platform.cmake)
 endif()
endif()

  fruit_warnings(fruit_platform_tests)
  add_test(NAME fruit_platform_tests COMMAND fruit_platform_tests "${CMAKE_CURRENT_SOURCE_DIR}/assets")
  add_test(NAME fruit_headless_smoke COMMAND fruit_ninja --headless --mute --frames 360 --assets "${CMAKE_CURRENT_SOURCE_DIR}/assets" --save-dir "${CMAKE_CURRENT_BINARY_DIR}/test-saves")
 endif()
