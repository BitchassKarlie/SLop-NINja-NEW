cmake_minimum_required(VERSION 3.16)
project(fruit_asset_recovery LANGUAGES CXX)
add_library(fruit_assets STATIC src/fruit_assets.cpp)
target_include_directories(fruit_assets PUBLIC include)
target_compile_features(fruit_assets PUBLIC cxx_std_17)
option(FRUIT_BUILD_INSPECTOR "Build desktop asset inspection CLI" ON)
if(FRUIT_BUILD_INSPECTOR)
 add_executable(fruit_inspect src/inspect.cpp)
 target_link_libraries(fruit_inspect PRIVATE fruit_assets)
endif()
